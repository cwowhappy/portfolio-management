# P3 · 估值数据采集服务 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增 Python 独立采集服务 `collector/`：每日收盘后经 akshare + tushare 采集全 A 股估值与国债收益率、指数 PE/PB 历史、申万行业映射，用 pandas 计算中位数（剔除口径）/破净/行业加权，经 psycopg 幂等写入 P1 定义的 PostgreSQL 表，APScheduler 调度。

**Architecture:** 采集（写）与查询（读）分离：本服务只做 INSERT/UPDATE（无 DDL），写入共享 PostgreSQL 中由 Flyway 定义的 `valuation_snapshot` / `treasury_yield` / `index_valuation_history` / `shenwan_industry_mapping` / `industry_valuation` 五表。分层 `sources`（akshare/tushare adapter，归一化列名）→ `calc`（pandas 纯计算）→ `store`（psycopg 幂等 upsert）→ `scheduler`（APScheduler）→ `backfill`（指数历史回填 runner）。

**Tech Stack:** Python 3.12 / pandas / akshare / tushare / APScheduler / psycopg[binary] / python-dotenv / pytest / pytest-cov · PostgreSQL 16

**Spec:** [需求规格说明](../01-requirement/需求规格说明.md)（§三 FR-A1/A2/A3/A4）· [ADR-0003 行情数据源](../../../docs/technology/decisions/0003-market-data-source.md) · [数据需求附录](../../../docs/function/99-附录-数据需求与技术扩展.md)

## Global Constraints

- **schema 由 Flyway（Java）单一拥有**（P1 的 `V3__valuation.sql`）；本服务**不建表、不迁移**，只 INSERT/UPDATE，表名与列名逐字对齐 P1。
- 跨服务口径：全 A PE/PB 中位数剔除 PE≤0 与 PE>100；快照按交易日唯一（`ON CONFLICT (trading_day) DO UPDATE` 幂等）。
- 数据源：akshare（免费无 token，全 A 估值 + 国债）、tushare（需 `TUSHARE_TOKEN`，指数 PE/PB 历史 + 申万映射）；adapter 一律经接口隔离，测试用 mock/fixtures 不碰真实网络。
- 调度：APScheduler cron 触发器，交易日收盘后（`0 30 15 * * 1-5`，即交易日 15:30）。
- 测试：pytest + coverage ≥ 80%（与 Java 门槛对齐）。
- 本服务不提供任何 HTTP API；数据经共享 PostgreSQL 交付给后端读侧。

`calc/snapshot.py`：

```python
import pandas as pd

def compute_snapshot(universe: pd.DataFrame) -> dict:
    pe = universe["pe"]
    valid = pe[(pe > 0) & (pe <= 100)]
    pb = universe["pb"].dropna()
    total = len(universe)
    net_breaker_count = int((universe["pb"] < 1).sum())
    return {
        "pe_median": float(valid.median()),
        "pb_median": float(pb.median()),
        "net_breaker_count": net_breaker_count,
        "net_breaker_ratio": round(net_breaker_count / total, 4) if total else 0.0,
    }

def compute_industry_valuation(universe: pd.DataFrame, mapping: pd.DataFrame) -> list[dict]:
    df = universe.merge(mapping, on="code", how="inner")
    df = df[df["pe"] > 0]  # 剔除亏损股（PE 不可比）
    rows = []
    for industry, g in df.groupby("industry_name"):
        cap = g["market_cap"].dropna()
        if cap.sum() <= 0:
            continue
        pe_w = float((g["pe"] * cap).sum() / cap.sum())
        pb_w = float((g["pb"] * cap).sum() / cap.sum())
        rows.append({"industry_name": industry, "pe": round(pe_w, 4), "pb": round(pb_w, 4)})
    return rows
```

> 说明：`industry_valuation.industry_code` 与 `roe`/`dividend_yield` 暂置空或由写入层以行业名兜底，`industry_code` 非空约束由写入层补默认值（如行业名）；ROE/股息率聚合留待数据源可用时扩展（SRS FR-B7 的展示列）。

`store/writer.py`：

```python
import datetime as dt
from typing import Iterable

import psycopg

SNAPSHOT_UPSERT = """
INSERT INTO valuation_snapshot (trading_day, pe_median, pb_median, net_breaker_count, net_breaker_ratio)
VALUES (%s, %s, %s, %s, %s)
ON CONFLICT (trading_day) DO UPDATE SET
  pe_median = EXCLUDED.pe_median,
  pb_median = EXCLUDED.pb_median,
  net_breaker_count = EXCLUDED.net_breaker_count,
  net_breaker_ratio = EXCLUDED.net_breaker_ratio
"""

TREASURY_UPSERT = """
INSERT INTO treasury_yield (trading_day, yield_10y)
VALUES (%s, %s)
ON CONFLICT (trading_day) DO UPDATE SET yield_10y = EXCLUDED.yield_10y
"""

INDEX_UPSERT = """
INSERT INTO index_valuation_history (trading_day, index_code, index_name, pe, pb, dividend_yield)
VALUES (%s, %s, %s, %s, %s, %s)
ON CONFLICT (trading_day, index_code) DO UPDATE SET
  pe = EXCLUDED.pe, pb = EXCLUDED.pb, dividend_yield = EXCLUDED.dividend_yield
"""

INDUSTRY_UPSERT = """
INSERT INTO industry_valuation (trading_day, industry_code, industry_name, pe, pb, roe, dividend_yield)
VALUES (%s, %s, %s, %s, %s, NULL, NULL)
ON CONFLICT (trading_day, industry_code) DO UPDATE SET
  pe = EXCLUDED.pe, pb = EXCLUDED.pb
"""

MAPPING_UPSERT = """
INSERT INTO shenwan_industry_mapping (stock_code, stock_name, industry_code, industry_name)
VALUES (%s, %s, %s, %s)
ON CONFLICT (stock_code) DO UPDATE SET
  industry_code = EXCLUDED.industry_code, industry_name = EXCLUDED.industry_name
"""

def upsert_snapshot(conn, trading_day: dt.date, snap: dict) -> None:
    with conn.cursor() as cur:
        cur.execute(SNAPSHOT_UPSERT, (trading_day, snap["pe_median"], snap["pb_median"],
                                      snap["net_breaker_count"], snap["net_breaker_ratio"]))
    conn.commit()

def upsert_treasury(conn, trading_day: dt.date, yield_10y: float) -> None:
    with conn.cursor() as cur:
        cur.execute(TREASURY_UPSERT, (trading_day, yield_10y))
    conn.commit()

def upsert_index(conn, trading_day, index_code: str, index_name: str, pe, pb, dividend_yield) -> None:
    with conn.cursor() as cur:
        cur.execute(INDEX_UPSERT, (trading_day, index_code, index_name, pe, pb, dividend_yield))
    conn.commit()

def upsert_industry(conn, trading_day: dt.date, rows: Iterable[dict]) -> None:
    with conn.cursor() as cur:
        for r in rows:
            code = r.get("industry_code") or r["industry_name"]
            cur.execute(INDUSTRY_UPSERT, (trading_day, code, r["industry_name"], r["pe"], r["pb"]))
    conn.commit()

def upsert_mapping(conn, rows: Iterable[tuple]) -> None:
    with conn.cursor() as cur:
        for stock_code, stock_name, industry_code, industry_name in rows:
            cur.execute(MAPPING_UPSERT, (stock_code, stock_name, industry_code, industry_name))
    conn.commit()
```

`run_once.py`（同时作为 `collect_once` 的入口）：

```python
import datetime as dt

import psycopg
import tushare as ts

from collector.config import load
from collector.sources.universe import fetch_all_a_valuation
from collector.sources.treasury import fetch_treasury_10y
from collector.sources.industry import fetch_shenwan_mapping
from collector.calc.snapshot import compute_snapshot, compute_industry_valuation
from collector.store.writer import upsert_snapshot, upsert_treasury, upsert_industry, upsert_mapping


def collect_once(conn, config) -> None:
    today = dt.date.today()
    universe = fetch_all_a_valuation()
    snapshot = compute_snapshot(universe)

    pro = ts.pro_api(config.tushare_token)
    mapping = fetch_shenwan_mapping(pro)
    industry_rows = compute_industry_valuation(universe, mapping)

    upsert_snapshot(conn, today, snapshot)
    upsert_treasury(conn, today, fetch_treasury_10y())
    upsert_industry(conn, today, industry_rows)
    upsert_mapping(conn, [(r.code, r.name, r.industry_name, r.industry_name) for r in mapping.itertuples()])


if __name__ == "__main__":
    config = load()
    with psycopg.connect(config.database_url) as conn:
        collect_once(conn, config)
        print("采集完成")
```

```python
from apscheduler.schedulers.blocking import BlockingScheduler
from apscheduler.triggers.cron import CronTrigger

import psycopg

from collector.config import load
from collector.run_once import collect_once


def main() -> None:
    config = load()
    scheduler = BlockingScheduler()
    scheduler.add_job(
        lambda: _run(config),
        CronTrigger(day_of_week="mon-fri", hour=15, minute=30),
        id="valuation_daily_snapshot",
        coalesce=True,
    )
    scheduler.start()


def _run(config) -> None:
    with psycopg.connect(config.database_url) as conn:
        collect_once(conn, config)


if __name__ == "__main__":
    main()
```
